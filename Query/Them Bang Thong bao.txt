-- ================================================
-- 10. BẢNG THONG_BAO
-- ================================================
CREATE TABLE ThongBao (
    MaThongBao SERIAL PRIMARY KEY,
    TieuDe VARCHAR(255) NOT NULL,
    NoiDung TEXT NOT NULL,
    MaTepDinhKem INT REFERENCES TepDinhKem(MaTepDinhKem) ON DELETE SET NULL,
    NguoiDang CHAR(12) REFERENCES NguoiDung(CCCD) ON DELETE SET NULL,
    ThoiGianDang TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    TrangThai VARCHAR(20) DEFAULT 'HienThi'
        CHECK (TrangThai IN ('HienThi', 'An'))
);

COMMENT ON TABLE ThongBao IS 
    'Lưu thông báo do cán bộ hoặc quản lý đăng lên trang chủ hệ thống.';

COMMENT ON COLUMN ThongBao.MaTepDinhKem IS 
    'Tệp đính kèm đi kèm thông báo (nếu có).';
